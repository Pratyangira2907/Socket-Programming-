TCP Server

tcpserver.py · Listens for a client and converts Celsius to Fahrenheit
import socket

HOST = "127.0.0.1"
PORT = 65432

# Create TCP socket
server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Bind IP and port
server.bind((HOST, PORT))

# Listen for client
server.listen()
print("Waiting for client...")

# Accept connection
conn, addr = server.accept()
print("Connected by", addr)

# Receive Celsius value
data = conn.recv(1024).decode()
celsius = float(data)

# Convert Celsius to Fahrenheit
fahrenheit = (celsius * 9/5) + 32

# Send result to client
result = str(fahrenheit)
conn.send(result.encode())

print("Celsius:", celsius)
print("Fahrenheit:", fahrenheit)

conn.close()
server.close()
Run this file first: python tcpserver.py

TCP Client

tcpclient.py · Sends Celsius to the server and displays the reply
import socket

HOST = "127.0.0.1"
PORT = 65432

# Create TCP socket
client = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Connect to server
client.connect((HOST, PORT))

# Get Celsius temperature
celsius = input("Enter temperature in Celsius: ")

# Send Celsius value
client.send(celsius.encode())

# Receive Fahrenheit result
data = client.recv(1024).decode()

print("Temperature in Fahrenheit:", data)

client.close()
Run this after starting the TCP server: python tcpclient.py

UDP Server

udpserver.py · Receives a datagram and sends the converted temperature
import socket

# Create UDP socket
server = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

HOST = "127.0.0.1"
PORT = 12345

# Bind IP and port
server.bind((HOST, PORT))

print("Waiting for client...")

# Receive Celsius value
data, addr = server.recvfrom(1024)

celsius = float(data.decode())

# Convert Celsius to Fahrenheit
fahrenheit = (celsius * 9/5) + 32

# Send result to client
result = str(fahrenheit)
server.sendto(result.encode(), addr)

print("Celsius:", celsius)
print("Fahrenheit:", fahrenheit)

server.close()
Run this file first: python udpserver.py

UDP Client

udpclient.py · Sends Celsius to the UDP server and displays the reply
import socket

# Create UDP socket
client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

HOST = "127.0.0.1"
PORT = 12345

# Get Celsius temperature
celsius = input("Enter temperature in Celsius: ")

# Send Celsius value to server
client.sendto(celsius.encode(), (HOST, PORT))

# Receive Fahrenheit result
data, addr = client.recvfrom(1024)

print("Temperature in Fahrenheit:", data.decode())

client.close()
Run this after starting the UDP server: python udpclient.py
